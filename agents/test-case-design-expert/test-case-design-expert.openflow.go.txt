package main

import (
	"strings"

	. "github.com/xhd2015/openflow/sdk"
)

func main() {
	coder := NewAgent(AgentOpts{
		Name: "coder",
		SystemPrompt: `You are a Go programmer. Write clean, tested code. Follow the exact spec given to you.

## Spec for main.go

Create the file main.go in the current directory. It must be a standalone CLI that takes a feature description and outputs user-facing e2e test cases.

### Behavior
- Read CLI args: positional args joined as the feature description, optional --model flag
- If no args: print usage to stderr and exit 1
- Call the LLM via this project's internal run.Run() to get test cases
- Print the result to stdout

### Imports to use
- "github.com/xhd2015/agent-pro/agent/opencode/run"
- "github.com/xhd2015/agent-pro/agent/opencode/models"

### LLM call pattern (same as commit_msg agent)
- Use run.Run(run.Options{Prompt: "...", Dir: ".", Model: model, Logger: &stdoutLogger{}})
- IMPORTANT: run.Logger is an interface with Log(string). You MUST pass a non-nil Logger (it is NOT optional — the function crashes on nil). Define a simple struct implementing run.Logger (e.g. stdoutLogger with a Log method that writes to os.Stderr or does nothing).
- Model defaults to the first free model from models.ListFree()
- The prompt must instruct the LLM: "You are a senior QA engineer. Given a feature description, design user-facing end-to-end test cases..." and include the user's feature description

### Output format
The LLM response should contain test cases with these sections for each:
- Scenario: what the user does
- Steps: concrete reproducible steps
- Expected: what should happen

### Error handling
- If run.Run fails, print error to stderr and exit 1
- If models.ListFree fails, print error to stderr and exit 1

## Spec for main_test.go

Create the file main_test.go in the same directory. It must test the compiled binary.

### Test cases
1. TestBuild: go build succeeds
2. TestRun: run the binary with "a login page with email and password authentication" and verify output contains keywords: "Scenario", "Steps", "Expected", "login"
3. TestNoArgs: run with no args, verify non-zero exit and stderr has "Usage"
4. TestModelFlag: run with --model flag, verify it doesn't crash

### Implementation
- Use os/exec to build and run the binary
- Build to a temp file (os.CreateTemp)
- Use testing.T assertions`,
	})

	reviewer := NewAgent(AgentOpts{
		Name: "reviewer",
		SystemPrompt: `You are a code reviewer. Check Go code against a spec. Be strict but fair.

The spec you are checking against:

### main.go requirements
- Package main with func main()
- Reads CLI args: positional = feature description, --model flag
- No args → prints usage to stderr, exit 1
- Imports run from "github.com/xhd2015/agent-pro/agent/opencode/run"
- Imports models from "github.com/xhd2015/agent-pro/agent/opencode/models"
- Calls run.Run(run.Options{...}) with the feature description as prompt AND a non-nil Logger (required, nil crashes)
- Uses models.ListFree() for default model
- Prints LLM output to stdout
- Handles errors properly

### main_test.go requirements
- TestBuild: builds the binary, verifies file exists
- TestRun: runs binary with sample input, asserts output contains "Scenario", "Steps", "Expected", "login"
- TestNoArgs: runs binary with no args, asserts exit != 0, stderr has "Usage"
- TestModelFlag: runs binary with --model flag, asserts exit == 0

### Review criteria
Respond with "PASS" if code matches spec and looks correct.
Respond with "FAIL: <reason>" if there are issues.`,
	})

	done := false
	i := 0
	const MAX = 5
	testCmd := "go test ./..."
	buildCmd := "go build -o /tmp/test-case-design-expert ."
	var reviewerFeedback string

	feedbacks := []Feedback{}

	for !done && i < MAX {
		i++
		Step("ITERATION "+S(i), func() {
			Step("CODING", func() {
				task := "Write main.go and main_test.go in the current directory according to the spec in your system prompt."
				if i > 1 {
					task = "The previous attempt had issues. Fix main.go and main_test.go. "
					if reviewerFeedback != "" && !strings.Contains(strings.ToLower(reviewerFeedback), "pass") {
						task += "Reviewer said: " + reviewerFeedback + " "
					}
					task += "Fix the tests so they pass."
				}
				_, _, err := coder.Run(task, RunOpts{Feedbacks: feedbacks})
				if err != nil {
					Print("Coder failed: " + err.Error())
					return
				}
			})

			iterFeedbacks := []Feedback{}

			Step("BUILD", func() {
				r := Shell(buildCmd, ShellOpts{Name: "build"})
				if r.ExitCode != 0 {
					Print("Build failed (exit " + S(r.ExitCode) + ")")
					iterFeedbacks = append(iterFeedbacks, r.Feedback)
					return
				}
				Print("Build passed")
			})

			if len(iterFeedbacks) > 0 {
				feedbacks = iterFeedbacks
				return
			}

			Step("REVIEW", func() {
				resp, _, err := reviewer.Run(
					"Review main.go and main_test.go against the spec. Read both files first.",
					RunOpts{},
				)
				if err != nil {
					Print("Reviewer failed: " + err.Error())
					return
				}
				reviewerFeedback = resp
				if strings.Contains(strings.ToLower(resp), "pass") && !strings.Contains(strings.ToLower(resp), "fail") {
					Print("Review passed")
				} else {
					Print("Review found issues: " + resp)
				}
			})

			Step("TEST", func() {
				r := Shell(testCmd, ShellOpts{Name: "test"})
				if r.ExitCode == 0 {
					done = true
					Print("All tests pass!")
					return
				}
				Print("Tests failed (exit " + S(r.ExitCode) + ")")
				iterFeedbacks = append(iterFeedbacks, r.Feedback)
			})

			feedbacks = iterFeedbacks
		})
	}

	if done {
		Print("SUCCESS — main.go and main_test.go created and verified.")
	} else {
		Print("GAVE UP after " + S(MAX) + " iterations.")
	}
}
